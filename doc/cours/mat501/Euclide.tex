\bigskip
\centerline{\Large\bf Géométrie d'Euclide}
\bigskip


{\bf Définitions}
\begin{description}
\item[1)] Un \emph{point} est un objet qui n'a pas de dimension.

\item[2)] Une \emph{droite} est un objet qui a une longueur, mais pas de largeur. Pour Euclide, la droite est notre segment de droite.

\item[3)] Une \emph{surface} est un objet qui a deux dimensions.

\item[4)] Un \emph{angle} est la donnée de deux droites qui ont un point en commun.

\item[5)] Un angle $\alpha$ est \emph{droit} s'il est égal à son complémentaire (on note le complémentaire d'un angle $\alpha$ par $\alpha'$).

\item[6)] Un \emph{cercle} est le lieu des points situés à même distance d'un point \mbox{donné $C$}.

\item[7)] Deux droites $d$ et $d'$ sont \emph{parallèles} si elles n'ont aucun point en commun si on les prolonge indéfiniment.

\end{description}

\vspace*{2cm}

{\bf Notions communes : axiomes logiques}

\begin{description}

\item[N1)] Si $A=B$ et $C=B$, alors $A=C$.

\item[N2)] Si $A=B$, alors $A+C = B+C$.

\item[N3)] Si $A=B$, alors $A-C = B-C$.

\item[N4)] Si $A \cong B$, alors $A=B$. (On dit que $A$ est congruent à $B$, noté $A \cong B$, si on peut placer l'un sur l'autre et qu'alors ils se confondent.)

\item[N5)] Le tout est toujours plus grand qu'une partie.

\end{description}



\newpage

{\bf Postulats : axiomes géométriques}

\begin{description}

\item[P1)] Étant donnés deux points $A$ et $B$, il existe une et une seule droite $d$ de $A$ à $B$ et on note $d=(A,B)$.

\item[P2)] Étant donné une droite $d=(A,B)$, il est possible de prolonger infiniment $d$ sur chaque côté.

\item[P3)] Étant donné un point $C$ et une droite $d=(A,B)$, il existe un cercle unique $\Gamma_C$ de centre $C$ et de rayon $(A,B)$.

\item[P4)] Tous les angles droits sont égaux.

\item[P5)] Soient $d$ et $d'$ deux droites et $t$ une droite
transversale avec les angles $\alpha$ et $\alpha'$ entre les
droites $d$ et $d'$, mais du même côté de $t$ (angles internes).
Si $\alpha + \alpha'$ est plus petit que deux angles droits, alors
il existe un point d'intersection $P$ de $d$ et $d'$ si $d$ et
$d'$ sont suffisamment prolongées.

\end{description}
