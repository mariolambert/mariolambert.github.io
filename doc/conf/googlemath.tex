\begin{frame}
  \frametitle{Qu'est-ce qu'un moteur de recherche ?}

  \begin{definition}
  Un \alert{moteur de recherche}, c'est un outil qui accomplit essentiellement trois
  tâches :
    \begin{itemize}
    \item parcourir le Web à la recherche des pages publiques ;
    \item indexer ces pages de façon à pouvoir les fouiller
    intelligemment pour des mots-clés ou des phrases ;
    \item ordonner les pages par ordre d'importance
    \end{itemize}
  \end{definition}

\end{frame}

\section{La bête}


\begin{frame}[fragile]
  \frametitle{La base}
  \begin{itemize}
  \item<1-> L'importance d'une page $P_i$ est définie par

  $$I(P_i) = \sum\limits_{\overset{P_j \neq P_i}{P_j \to P_i}}
  \frac{I(P_j)}{l_j}$$

  où $l_j$ est le nombre de liens sortant de la page $P_j$.

  \item<2-> La matrice de Google est $H = [H_{ij}]$ où

  $$H_{ij}= \left\lbrace
  \begin{array}{cl}
  \frac{1}{l_j} & \textrm{si } P_i \to P_j \\\\
  0 & \textrm{sinon.}
  \end{array}
  \right.$$
  \end{itemize}
\end{frame}


\begin{frame}[fragile]
  \frametitle{Formulation en terme de vecteur propre}
  $$\begin{array}{rcl}
  I(P_i) & = & \sum\limits_{\overset{P_j \neq P_i}{P_j \to P_i}} \frac{I(P_j)}{l_j} \\\\
         & = & \sum\limits_{j \neq i} I(P_j) H_{ji} \\\\
         & = & H_{1i}I(P_1) + H_{2i}I(P_2) + \ldots + H_{ni}I(P_n) \\\\
         & = & [H^tI]_i
  \end{array}$$

  où $I$ est le vecteur formé des importance de toutes les pages du Web.
\end{frame}


\begin{frame}[fragile]
  \frametitle{Premier problème}
Est-ce que $H^t$ admet toujours 1 comme valeur propre?
\bigskip

\begin{theorem}
Si la somme des entrées sur chaque ligne de $H$ est 1, alors $H^t$ a 1 comme valeur propre.
\end{theorem}

\end{frame}
\begin{frame}[fragile]
  \frametitle{Deuxième problème}
Si 1 est une valeur propre de $H^t$, est-ce que le vecteur $I$ est uniquement déterminé ?

\begin{theorem}
Si $\overline{H}$ est strictement positive et que la somme des entrées sur chaque ligne de $\overline{H}$ est 1, alors l'espace propre associé à la valeur propre 1 est de dimension 1.
\end{theorem}

\end{frame}






\section{La bête numérique}
\begin{frame}[fragile]
  \frametitle{Algorithme}
  $A$ une matrice carrée.

  Les valeurs propres de $A$ sont $1=\lambda_1 > |\lambda_2| \geqslant |\lambda_3| \geqslant \ldots \geqslant |\lambda_n|$.

  Les vecteurs propres de $A$ sont $x_1, x_2, \ldots x_n$.

  Supposons que les vecteurs propres de $A$ forment une base de $\mathbb{R}^n$.

  Soit $I^{(0)}$ un vecteur quelconque non nul.

  Alors, $I^{(0)} = \alpha_1x_1 + \alpha_2x_2 + \ldots + \alpha_n x_n$.
  Posons $I^{(k)} = AI^{(k-1)}$.

  \begin{theorem}
    $I^{(k)} \to x_1$
  \end{theorem}
\end{frame}
\begin{frame}[fragile]
  \frametitle{Premier problème}
  Est-ce que dans le cas de $\overline{\overline{H}}$, $$1=\lambda_1 > |\lambda_2| \geqslant |\lambda_3| \geqslant \ldots \geqslant |\lambda_n|$$
\end{frame}
\begin{frame}[fragile]
  \frametitle{Vitesse de convergence}
    \begin{theorem}
    $\lambda_2 = \alpha$
    \end{theorem}

\end{frame}
\begin{frame}[fragile]
  \frametitle{Variations sur un même thème}
  \begin{itemize}
  \item<1-> $\alpha = 0,85$
  \item<2-> $\alpha = 0,9999$
  \item<3-> $\alpha = 0,0001$
  \end{itemize}
\end{frame}



\section{La bête stochastique}
\begin{frame}[fragile]
  \frametitle{Les équivalences}
  \begin{itemize}
    \item<1-> Web = \alert{chaîne de Markov}
    \item<2-> $H$ = \alert{matrice de transition}
    \item<3-> $\overline{H}$ est \alert{stochastique}
    \item<4-> Le vecteur $I$ des rangs est une \alert{distribution limite}
    \item<5-> Les composantes de $I$ représentent la \alert{proportion du temps} où un promeneur sur le Web est dans une page donnée
    \item<6-> connexité forte = \alert{irréductibilité} de la chaîne de Markov
    \item<7-> Si 1 est une valeur propre multiple, il y aura une page qui va bouffer tous les rangs, et c'est un \alert{état absorbant} de la chaîne de Markov.
  \end{itemize}
\end{frame}



\section{La bête algébrique}
\begin{frame}[fragile]
  \frametitle{Réécriture du problème}

\end{frame}
\begin{frame}[fragile]
  \frametitle{Est-ce que ça marche?}
    \begin{theorem}
    \begin{itemize}
    \item<1-> $1\!{\rm l} - \alpha \overline{H}$ est non singulière
    \item<2-> La somme des éléments sur une ligne de $1\!{\rm l} - \alpha \overline{H}$ est $1-\alpha$.
    \item<3-> $\Vert 1\!{\rm l} - \alpha \overline{H} \Vert_{\infty} = 1+\alpha$
    \item<4-> $\Vert \left( 1\!{\rm l} - \alpha \overline{H}\right)^{-1} \Vert_{\infty} = \frac{1}{1-\alpha}$
    \item<5-> ${\rm cond}(1\!{\rm l} - \alpha \overline{H}) = \frac{1+\alpha}{1-\alpha}$
    \end{itemize}
    \end{theorem}

\end{frame}

  \end{itemize}
